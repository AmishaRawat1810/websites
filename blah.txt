  background: radial-gradient(circle at 0% 10%, hsl(167 100% 43%), hsl(192 48% 26%) 30%, hsl(184 58% 42%), hsl(200 78% 37%));

  /* 2. Clip the background to the text */

  font-weight: bold;

}
